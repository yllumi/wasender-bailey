📨 How to Send Messages

Use your session to send WhatsApp messages via API

Endpoint: POST /{session_name}/send
Parameters:
  • app_key - Your API key (query parameter or X-App-Key header)
  • number - Recipient phone number (e.g., 628986818780)
  • message - Text message to send
Bash / Terminal 
curl -X POST "http://your-server:8990/yoursession/send?app_key=YOUR_APP_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "number": "628986818780",
    "message": "Hello from WhatsApp API!"
  }'
PHP 
<?php
$session = 'yoursession';
$url = "http://your-server:8990/{$session}/send?app_key=YOUR_APP_KEY";

$data = [
    'number' => '628986818780',
    'message' => 'Hello from WhatsApp API!'
];

$options = [
    'http' => [
        'header'  => "Content-Type: application/json\r\n",
        'method'  => 'POST',
        'content' => json_encode($data)
    ]
];

$context  = stream_context_create($options);
$result = file_get_contents($url, false, $context);

if ($result === FALSE) {
    die('Error sending message');
}

$response = json_decode($result, true);
print_r($response);
?>
JavaScript (Fetch API) 
async function sendWhatsAppMessage(sessionName, phoneNumber, message) {
  const url = `http://your-server:8990/${sessionName}/send?app_key=YOUR_APP_KEY`;
  
  try {
    const response = await fetch(url, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        number: phoneNumber,
        message: message
      })
    });
    
    const data = await response.json();
    
    if (data.success) {
      console.log('Message sent successfully!');
      return data;
    } else {
      console.error('Error:', data.message);
      throw new Error(data.message);
    }
  } catch (error) {
    console.error('Failed to send message:', error);
    throw error;
  }
}

// Usage example
sendWhatsAppMessage('yoursession', '628986818780', 'Hello from WhatsApp API!')
  .then(result => console.log('Success:', result))
  .catch(error => console.error('Error:', error));
⚠️ Important:
  • Replace yoursession with your session name
  • Phone numbers should include country code (e.g., 62 for Indonesia)
